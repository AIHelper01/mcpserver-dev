import java.io.File;
import java.io.FileWriter;
import java.io.IOException;

public class Main {
    public static void main(String[] args) {
        // 创建文件并写入内容
        try {
            File file = new File("test.txt");
            if (file.createNewFile()) {
                FileWriter writer = new FileWriter(file);
                writer.write("Hello, Java!");
                writer.close();
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
